# ============================================================ # AI STOCK PRICE PREDICTION SYSTEM # Google Colab + Machine Learning + Gradio UI # ============================================================ # STEP 1: INSTALL LIBRARIES !pip -q install yfinance gradio plotly scikit-learn pandas numpy # STEP 2: IMPORT LIBRARIES import gradio as gr import yfinance as yf import pandas as pd import numpy as np import plotly.graph_objects as go from sklearn.ensemble import RandomForestRegressor from sklearn.metrics import mean_absolute_error from datetime import datetime import warnings warnings.filterwarnings("ignore") # STEP 3: CREATE PREDICTION FUNCTION def predict_stock(ticker): try: ticker = ticker.strip().upper() if not ticker: raise ValueError("Please enter a stock ticker.") # Fetch historical dataset using Yahoo Finance data = yf.download( ticker, period="5y", interval="1d", auto_adjust=True, progress=False ) if data.empty: raise ValueError( "No stock data found. Check the ticker symbol." ) # Handle possible MultiIndex columns if isinstance(data.columns, pd.MultiIndex): data.columns = data.columns.get_level_values(0) if "Close" not in data.columns: raise ValueError("Closing price data is unavailable.") df = data[["Close"]].copy() # STEP 4: FEATURE ENGINEERING for lag in [1, 2, 3, 5, 10, 20]: df[f"Lag_{lag}"] = df["Close"].shift(lag) df["MA_5"] = df["Close"].rolling(5).mean() df["MA_10"] = df["Close"].rolling(10).mean() df["MA_20"] = df["Close"].rolling(20).mean() df["Daily_Return"] = df["Close"].pct_change() df["Volatility"] = ( df["Daily_Return"].rolling(10).std() ) # Predict next day's closing price df["Target"] = df["Close"].shift(-1) df = df.dropna() if len(df) < 100: raise ValueError( "Not enough historical data for training." ) features = [ "Lag_1", "Lag_2", "Lag_3", "Lag_5", "Lag_10", "Lag_20", "MA_5", "MA_10", "MA_20", "Daily_Return", "Volatility" ] X = df[features] y = df["Target"] # STEP 5: TRAIN-TEST SPLIT split = int(len(df) * 0.80) X_train = X.iloc[:split] X_test = X.iloc[split:] y_train = y.iloc[:split] y_test = y.iloc[split:] # STEP 6: TRAIN MACHINE LEARNING MODEL model = RandomForestRegressor( n_estimators=150, max_depth=10, random_state=42, n_jobs=-1 ) model.fit(X_train, y_train) # STEP 7: MODEL EVALUATION test_predictions = model.predict(X_test) mae = mean_absolute_error( y_test, test_predictions ) # STEP 8: PREDICT NEXT TRADING DAY latest_features = df[features].iloc[[-1]] predicted_price = model.predict( latest_features )[0] current_price = float(data["Close"].iloc[-1]) difference = predicted_price - current_price percentage = ( difference / current_price ) * 100 last_date = data.index[-1].strftime("%d %b %Y") # STEP 9: CREATE DASHBOARD GRAPH fig = go.Figure() fig.add_trace( go.Scatter( x=data.index[-180:], y=data["Close"].iloc[-180:], mode="lines", name="Historical Price", line=dict(width=2) ) ) next_date = data.index[-1] + pd.offsets.BDay(1) fig.add_trace( go.Scatter( x=[data.index[-1], next_date], y=[current_price, predicted_price], mode="lines+markers", name="Illustrative Prediction", line=dict(dash="dash", width=3) ) ) fig.update_layout( title=f"{ticker} Stock Price Analysis", xaxis_title="Date", yaxis_title="Stock Price", template="plotly_dark", height=450, hovermode="x unified" ) # STEP 10: DISPLAY RESULTS result = f""" # 📊 AI Stock Prediction Results ### Stock Symbol: {ticker} | Parameter | Result | |---|---| | Last Available Date | {last_date} | | Latest Closing Price | {current_price:.2f} | | Predicted Next Close | {predicted_price:.2f} | | Expected Change | {difference:.2f} | | Expected Change % | {percentage:.2f}% | | Model MAE | {mae:.2f} | ### 🤖 Machine Learning Model - Algorithm: Random Forest Regressor - Training Data: 80% - Testing Data: 20% - Historical Data: 5 Years - Evaluation Metric: Mean Absolute Error ### Prediction Status {"Higher than last close" if difference > 0 else "Lower than last close" if difference < 0 else "Unchanged"} ⚠️ Educational prediction only. This model is not a reliable financial forecasting tool or investment advice. """ # Recent historical data history = data.tail(10).reset_index() history.columns = [ "Date", "Closing Price" ] history["Date"] = ( history["Date"].dt.strftime("%Y-%m-%d") ) return result, fig, history except Exception as e: error = f""" ### ❌ Error {str(e)} Please check the stock ticker symbol and try again. """ return error, None, pd.DataFrame() # STEP 11: CREATE PROFESSIONAL GRADIO UI custom_css = """ .gradio-container { background: linear-gradient(135deg, #101827, #172554); } #main-title { text-align: center; color: #38bdf8; } #subtitle { text-align: center; color: #cbd5e1; } """ with gr.Blocks( theme=gr.themes.Soft( primary_hue="blue", secondary_hue="cyan" ), css=custom_css, title="AI Stock Price Prediction" ) as demo: gr.Markdown( """ # 📈 AI STOCK PRICE PREDICTION SYSTEM """, elem_id="main-title" ) gr.Markdown( """ ### Machine Learning Powered Stock Market Analysis Analyze historical stock data and generate an experimental next-day price prediction. """, elem_id="subtitle" ) gr.Markdown("---") with gr.Row(): stock_input = gr.Textbox( label="Enter Stock Symbol", placeholder="Example: AAPL, MSFT, RELIANCE.NS", value="AAPL", scale=3 ) predict_button = gr.Button( "🚀 Predict Stock Price", variant="primary", scale=1 ) gr.Markdown("### 📊 Prediction Dashboard") result_output = gr.Markdown( "Enter a stock symbol and click Predict." ) graph_output = gr.Plot( label="Stock Price Chart" ) gr.Markdown("### 📅 Recent Stock History") table_output = gr.Dataframe( label="Historical Stock Data", interactive=False ) predict_button.click( fn=predict_stock, inputs=stock_input, outputs=[ result_output, graph_output, table_output ] ) stock_input.submit( fn=predict_stock, inputs=stock_input, outputs=[ result_output, graph_output, table_output ] ) gr.Markdown( """ --- ### 🧠 About This Project This project uses Machine Learning to analyze historical stock market data. **Technology:** Python | Pandas | NumPy | Scikit-learn | Yahoo Finance | Plotly | Gradio **Disclaimer:** Predictions are experimental educational outputs, not financial advice. """ ) # STEP 12: LAUNCH PUBLIC WEB APPLICATION print("🚀 Starting AI Stock Prediction Application...") demo.launch( share=True, debug=True )
